คู่มือการใช้งานระบบ API ค่าส่วนกลาง (Common Fee API)

เอกสารประกอบสำหรับแอปพลิเคชันภายนอกในการเชื่อมโยงข้อมูลยอดค้างชำระค่าส่วนกลางของโครงการ PSHOME33

รายละเอียด Endpoint

ระบบอนุญาตให้แอปพลิเคชันภายนอกสามารถเรียกตรวจสอบยอดค้างชำระค่าส่วนกลางของบ้านแต่ละหลังผ่านโปรโตคอล HTTP/HTTPS ด้วยพารามิเตอร์ของบ้านเลขที่และปีบัญชี

GET /api/get_outstanding_fee.php

การรักษาความปลอดภัย (Authentication): จำเป็นต้องแนบคีย์ความปลอดภัย (API Key) ผ่าน Query Parameter หรือ HTTP Header ด้านล่างนี้:

  • API Key: PSHOME33_SECURE_API_KEY_2026
  • ชื่อ HTTP Header: X-API-KEY

พารามิเตอร์ของการเรียกใช้ (Request Parameters)

ชื่อ Parameter ประเภท ความจำเป็น รายละเอียด / ตัวอย่าง
api_key string จำเป็น * รหัสคีย์ความปลอดภัย PSHOME33_SECURE_API_KEY_2026 (สามารถส่งเป็น Header X-API-KEY แทนได้)
house_number string จำเป็น * บ้านเลขที่ที่ต้องการตรวจสอบ เช่น 67/117 หรือ 68/157
year int เลือกระบุได้ ปี ค.ศ. ที่ต้องการตรวจสอบยอด เช่น 2025 (หากไม่ระบุจะตรวจด้วยปีปัจจุบันโดยอัตโนมัติ)

รูปแบบการส่งกลับข้อมูล (Response Format)

เมื่อค้นพบข้อมูลบ้าน เลขบัญชี และประวัติการจ่ายชำระ ระบบจะส่งคืนโครงสร้างข้อมูล JSON รายเดือนของปีที่ร้องขอดังนี้:

กรณีระบุรหัสความปลอดภัยผิด หรือส่งบ้านเลขที่ที่ไม่มีอยู่ในระบบ ทะเบียนไม่ทำงาน จะส่งข้อมูลประเภทนี้กลับไป:

{
    "status": "error",
    "message": "Not Found: House number not found or inactive."
}

ตัวอย่างโค้ดสำหรับเชื่อมต่อ (Integration Codes)

curl -X GET \
  -H "X-API-KEY: PSHOME33_SECURE_API_KEY_2026" \
  "http://your-domain/api/get_outstanding_fee.php?house_number=67/117&year=2025"
const apiKey = 'PSHOME33_SECURE_API_KEY_2026';
const houseNo = '67/117';
const year = 2025;

fetch(`api/get_outstanding_fee.php?house_number=${houseNo}&year=${year}`, {
    method: 'GET',
    headers: {
        'X-API-KEY': apiKey
    }
})
.then(response => response.json())
.then(data => console.log(data))
.catch(error => console.error('Error:', error));
<?php
$apiKey = 'PSHOME33_SECURE_API_KEY_2026';
$houseNo = '67/117';
$year = 2025;

$url = "http://your-domain/api/get_outstanding_fee.php?house_number=" . urlencode($houseNo) . "&year=" . $year;

$ch = curl_init();
curl_setopt($ch, CURLOPT_URL, $url);
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
curl_setopt($ch, CURLOPT_HTTPHEADER, [
    "X-API-KEY: $apiKey"
]);

$response = curl_exec($ch);
curl_close($ch);

$data = json_decode($response, true);
print_r($data);
?>
