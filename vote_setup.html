คู่มือการพัฒนาและสถาปัตยกรรมข้อมูล (Integration Guide)

1. แนวคิดและตรรกะการตรวจสอบ (Validation Logic)

ในการจำกัดสิทธิ์ "1 บ้านเลขที่ ต่อ 1 โหวตในแต่ละหัวข้อ" ระบบจะทำการตรวจสอบแบบ Composite Key ที่ประกอบด้วย รหัสหัวข้อ (Topic ID) + บ้านเลขที่ (House Number) โดยมีลอจิกการทำงานดังนี้:

กระบวนการตรวจสอบ (Validation Step):
  1. รับค่าบ้านเลขที่ที่กรอกเข้ามา จากนั้นทำความสะอาดข้อมูล (Data Normalization) เช่น ตัดช่องว่างหัวท้าย, ปรับตัวอักษรให้อยู่ในรูปแบบเดียวกัน
  2. สืบค้นจากประวัติการลงคะแนน (ในฐานข้อมูล) ค้นหาว่ามีเรคคอร์ดที่ตรงกับ Topic ID และ House Number แล้วหรือไม่
  3. หากพบข้อมูล ระบบจะแสดงการแจ้งเตือนปฏิเสธการโหวต พร้อมชี้แจงวันเวลาที่เคยโหวตไปแล้ว
  4. หากไม่พบข้อมูล ระบบจะบันทึกผลการโหวตเข้าสู่ฐานข้อมูล
2. โครงสร้างตารางฐานข้อมูลที่แนะนำ (MySQL / PostgreSQL)

การพัฒนาบนฐานข้อมูล SQL แนะนำให้ออกแบบตารางออกเป็น 3 ตารางดังนี้:

1) ตารางหัวข้อโหวต (ims_vote_topic) 
CREATE TABLE `ims_vote_topic` (
  `topic_id` INT AUTO_INCREMENT PRIMARY KEY,
  `title` VARCHAR(255) NOT NULL COMMENT 'หัวข้อการโหวต',
  `description` TEXT NULL COMMENT 'รายละเอียดหัวข้อ',
  `status` ENUM('active', 'inactive') DEFAULT 'active' COMMENT 'สถานะการเปิดโหวต',
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
2) ตารางตัวเลือก (ims_vote_option) 
CREATE TABLE `ims_vote_option` (
  `option_id` INT AUTO_INCREMENT PRIMARY KEY,
  `topic_id` INT NOT NULL,
  `option_text` VARCHAR(255) NOT NULL COMMENT 'ข้อความตัวเลือก',
  FOREIGN KEY (`topic_id`) REFERENCES `ims_vote_topic`(`topic_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
3) ตารางเก็บประวัติการโหวต (ims_vote_record) 
CREATE TABLE `ims_vote_record` (
  `vote_id` INT AUTO_INCREMENT PRIMARY KEY,
  `topic_id` INT NOT NULL,
  `house_number` VARCHAR(50) NOT NULL COMMENT 'บ้านเลขที่ผู้โหวต',
  `option_id` INT NOT NULL COMMENT 'ไอดีตัวเลือกที่โหวต',
  `voted_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  /* ทำ UNIQUE KEY เพื่อบังคับระดับฐานข้อมูลให้ 1 บ้านเลขที่โหวตได้ 1 ครั้งในแต่ละหัวข้อ */
  UNIQUE KEY `uq_topic_house` (`topic_id`, `house_number`),
  FOREIGN KEY (`topic_id`) REFERENCES `ims_vote_topic`(`topic_id`) ON DELETE CASCADE,
  FOREIGN KEY (`option_id`) REFERENCES `ims_vote_option`(`option_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
3. ตัวอย่างการตรวจสอบระดับ Backend API (PHP Sample Code)

ตัวอย่างซอร์สโค้ด PHP ในการตรวจสอบและบันทึกคะแนนโหวตในฝั่งเซิร์ฟเวอร์:

<?php
// file: api_submit_vote.php
header('Content-Type: application/json');
require_once 'config.php'; // ไฟล์เชื่อมต่อฐานข้อมูล PDO

if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    echo json_encode(['status' => 'error', 'message' => 'Invalid Request Method']);
    exit;
}

$topic_id = filter_input(INPUT_POST, 'topic_id', FILTER_VALIDATE_INT);
$house_number = trim(filter_input(INPUT_POST, 'house_number', FILTER_UNSAFE_RAW));
$option_id = filter_input(INPUT_POST, 'option_id', FILTER_VALIDATE_INT);

if (!$topic_id || !$house_number || !$option_id) {
    echo json_encode(['status' => 'error', 'message' => 'กรุณากรอกข้อมูลให้ครบถ้วน']);
    exit;
}

try {
    // 1. ตรวจสอบว่าบ้านเลขที่นี้เคยโหวตหัวข้อนี้ไปแล้วหรือยัง
    $stmt = $pdo->prepare("SELECT voted_at FROM ims_vote_record WHERE topic_id = ? AND house_number = ?");
    $stmt->execute([$topic_id, $house_number]);
    $existing = $stmt->fetch();

    if ($existing) {
        echo json_encode([
            'status' => 'error', 
            'message' => "บ้านเลขที่ $house_number ได้ลงคะแนนในหัวข้อนี้ไปแล้วเมื่อ " . $existing['voted_at']
        ]);
        exit;
    }

    // 2. ทำการบันทึกคะแนนโหวต
    $insert = $pdo->prepare("INSERT INTO ims_vote_record (topic_id, house_number, option_id) VALUES (?, ?, ?)");
    $insert->execute([$topic_id, $house_number, $option_id]);

    echo json_encode([
        'status' => 'success',
        'message' => 'บันทึกการลงคะแนนของคุณเรียบร้อยแล้ว ขอบคุณสำหรับการมีส่วนร่วม'
    ]);

} catch (PDOException $e) {
    echo json_encode([
        'status' => 'error',
        'message' => 'เกิดข้อผิดพลาดในการบันทึกข้อมูล: ' . $e->getMessage()
    ]);
}
?>
4. การดึงและล็อกบ้านเลขที่อัตโนมัติจาก LINE LIFF (Auto-Locking House Number)

เพื่อป้องกันไม่ให้ลูกบ้านกรอกบ้านเลขที่อื่นหรือสวมรอยโหวตแทนบ้านอื่น เมื่อเปิดหน้าเว็บนี้ผ่าน LINE OA (LIFF) ระบบสามารถดึงประวัติการลงทะเบียนของ LINE Account นั้น และล็อก (Lock) ช่องบ้านเลขที่ให้เป็น readonly อัตโนมัติ:

กระบวนการล็อกสิทธิ์ (Flow):
  1. เรียกใช้งาน liff.init() และตรวจสอบการล็อกอินด้วย liff.isLoggedIn()
  2. เรียกคำสั่ง liff.getProfile() เพื่อดึง userId ของบัญชี LINE
  3. ยิง AJAX POST ไปยัง API หลังบ้านเพื่อดึงรายละเอียดบ้านเลขที่ที่เคยลงทะเบียนผูกไว้ในตาราง ims_house_line_user
  4. หากพบข้อมูล ระบบจะนำบ้านเลขที่มาใส่ในช่องกรอก และตั้งแอตทริบิวต์ readonly="true" ทันทีเพื่อล็อกค่าไม่ให้แก้ไขได้
  5. หากไม่พบการลงทะเบียนระบบจะแสดงป๊อปอัปแจ้งให้ลูกบ้านไปทำการลงทะเบียนผูกบัญชีก่อน จึงจะมีสิทธิ์เข้าถึงหน้าโหวตนี้

ตัวอย่างการเขียนโค้ดฝั่ง Javascript สำหรับเปิดใช้ระบบล็อกอัตโนมัติ:

// โค้ดเชื่อมต่อกับ LINE LIFF SDK
liff.init({ liffId: "YOUR_LIFF_ID" }).then(() => {
    if (liff.isLoggedIn()) {
        liff.getProfile().then(profile => {
            const lineUserId = profile.userId;
            
            // ยิงไปตรวจสอบที่หลังบ้านว่าเคยสมัครสมาชิกผูกเลขที่บ้านไว้หรือยัง
            $.post("https://ps33home.com/line_oa/house/register_house_with_line_api.php", {
                action: 'check',
                lineUserId: lineUserId
            }, function(data) {
                if (data.success && data.registered) {
                    const user = data.user;
                    
                    // 1. นำเลขที่บ้านที่ผูกกับ LINE มาใส่ในฟอร์มโดยอัตโนมัติ
                    const houseInput = document.getElementById('houseNumberInput');
                    houseInput.value = user.house_number;
                    
                    // 2. ทำการล็อก (Readonly) ช่องกรอกบ้านเลขที่ทันที ลูกบ้านจะไม่สามารถแก้ไขค่าได้
                    houseInput.setAttribute('readonly', 'true');
                    houseInput.style.backgroundColor = '#e9ecef'; // ทำเป็นสีเทาเพื่อแสดงว่าล็อกไว้
                    
                    console.log("ล็อกบ้านเลขที่สิทธิ์การโหวตเป็น: " + user.house_number);
                } else {
                    // หากยังไม่เคยผูกบัญชี LINE กับเลขที่บ้าน
                    alert("⚠️ บัญชี LINE นี้ยังไม่ได้ลงทะเบียนลูกบ้าน กรุณาลงทะเบียนก่อนเข้าโหวต");
                    window.location.href = "register_house_user.php";
                }
            });
        });
    } else {
        liff.login();
    }
}).catch(err => console.error("LIFF SDK Error:", err));
